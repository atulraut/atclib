 2D to 1D array Access : Using Binary Search
int n = row * col;
for (int i=0; i<n; ++i)
     debug("Output = %d", matrix[i/n][i%n]);

     ** OR **
 row = matrix[mid/n]
 col = matrix[mid%n]
 To Get Element from 1D Array -> matrix[mid/n][mid%n]
  e.g.
   Idx == Index in the array
   n   == Size of Column
   Idx: 0   1   3     |
        [1] [3] [5]   | n=3
        [6] [10][12]  | n=3
   Find 5 ->
   Row = idx/n (for 5, idx is 2 & n=3(no. of element in Row))
   Therefore Row = 2/3 = 0
   Col = idx % n = 2%3 = 2

    OR
bool searchMatrix2(int** matrix, int matrixSize, int* matrixColSize, int target){
  int i ,j;
  i = 0;
  j = *matrixColSize * matrixSize - 1;
  while( i <= j ){
    int mid = (i + j) / 2;
    int row ,col;
    row = mid / (*matrixColSize);
    col = mid % (*matrixColSize);
    if(matrix[row][col] < target)
      i = mid + 1;
    else if(matrix[row][col] > target)
      j = mid - 1;
    else
      return true;
  }
  return false;
}

void resize (int* qu, int N) {
   int *new;
   new = malloc (N * sizeof (int));
   for (int i = p; i < r; i++)
      new[i] = qu[i];
   free (qu);
   qu = new;
}
